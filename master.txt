def take_screenshot(path="screenshot.png"):
    img = pyautogui.screenshot()
    img.save(path)
    return path

def image_to_data_url(path):
    with open(path, "rb") as f:
        b64 = base64.b64encode(f.read()).decode("utf-8")
    return f"data:image/png;base64,{b64}"
